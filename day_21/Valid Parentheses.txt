public class Solution {
    public bool IsValid(string s) {
        Stack<char> stack = new Stack<char>();

        for(int i = 0;i<= s.Length -1 ;i++){
            if(s[i]=='(' || s[i]=='{' || s[i]=='['){
                stack.Push(s[i]);
            }
            else if(stack.Count > 0 && (s[i] == ')' && stack.Peek() == '(' ||s[i] == '}' && stack.Peek() == '{' || s[i] == ']' && stack.Peek() == '[')){
                stack.Pop();
            }
            else{
                return false;
            }
        }
        return true;
    }
        
}
