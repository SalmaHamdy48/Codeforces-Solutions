using System;
using System.Collections.Generic;

namespace myStack
{
    class Person
    {
        public int Id { get; set; }
        public string FName { get; set; }
        public string LName { get; set; }
        public string Gender { get; set; }
    }

    static class StackExtensions
    {
        
        public static string GetFirstPerson(this Stack<Person> stack, Predicate<Person> predicate)
        {
            foreach (var person in stack)
            {
                if (predicate(person))
                {
                    return $"ID: {person.Id}, Name: {person.FName} {person.LName}, Gender: {person.Gender}";
                }
            }
            return "No person found";
        }
    }

    class Program
    {
        static void Main(string[] args)
        {
            Stack<Person> stack = new Stack<Person>();

            stack.Push(new Person { Id = 1, FName = "Salma", LName = "Hamdy", Gender = "female" });
            stack.Push(new Person { Id = 2, FName = "Mohammed", LName = "Ahmed", Gender = "male" });
            stack.Push(new Person { Id = 3, FName = "Yehia", LName = "Hamdy", Gender = "male" });
            stack.Push(new Person { Id = 4, FName = "Mohammed", LName = "Hamdy", Gender = "male" });

            
            string result = stack.GetFirstPerson(p => p.FName == "Mohammed");

            Console.WriteLine("First Mohammed in the stack is: " + result);
        }
    }
}
